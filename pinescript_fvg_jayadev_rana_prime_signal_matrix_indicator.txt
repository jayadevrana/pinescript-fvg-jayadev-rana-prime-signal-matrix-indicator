var array<line> structureLines = array.new<line>()
var array<label> structureLabels = array.new<label>()
var array<box> zoneBoxes = array.new<box>()
var array<line> signalLines = array.new<line>()
var array<label> signalLabels = array.new<label>()

f_trim_structure() =>
    while array.size(structureLines) > maxStructureDrawings
        line.delete(array.shift(structureLines))
    while array.size(structureLabels) > maxStructureDrawings
        label.delete(array.shift(structureLabels))

f_trim_zones() =>
    while array.size(zoneBoxes) > maxZones
        box.delete(array.shift(zoneBoxes))

f_trim_signals() =>
    int maxLines = maxSignalDrawings * 4
    int maxLabels = maxSignalDrawings * 5
    while array.size(signalLines) > maxLines
        line.delete(array.shift(signalLines))
    while array.size(signalLabels) > maxLabels
        label.delete(array.shift(signalLabels))

f_add_structure_line(int startBar, float price, bool isBull, bool isChoch) =>
    if showStructure and not na(startBar)
        color lineColor = isBull ? buyColor : sellColor
        string tag = isChoch ? (isBull ? "CHoCH LONG" : "CHoCH SHORT") : (isBull ? "BOS LONG" : "BOS SHORT")
        line newLine = line.new(startBar, price, bar_index, price, color=color.new(lineColor, 10), width=1, style=line.style_dashed)
        label newLabel = label.new(bar_index, price, tag, style=isBull ? label.style_label_down : label.style_label_up, color=color.new(lineColor, 0), textcolor=color.white, size=size.tiny)
        array.push(structureLines, newLine)
        array.push(structureLabels, newLabel)
        f_trim_structure()

f_add_zone(bool isBull) =>
    if showFvgZones
        color zoneColor = isBull ? buyColor : sellColor
        float top = isBull ? low : low[2]
        float bottom = isBull ? high[2] : high
        box newBox = box.new(bar_index - 2, top, bar_index + zoneExtendBars, bottom, bgcolor=color.new(zoneColor, 88), border_color=color.new(zoneColor, 45))
        array.push(zoneBoxes, newBox)
        f_trim_zones()

f_stop_price(bool isBull) =>
    float atrStop = isBull ? close - atrValue * stopAtrMultiplier : close + atrValue * stopAtrMultiplier
    float swingStop = isBull and not na(lastSwingLow) and lastSwingLow < close ? lastSwingLow : not isBull and not na(lastSwingHigh) and lastSwingHigh > close ? lastSwingHigh : atrStop
    isBull ? math.max(swingStop, atrStop) : math.min(swingStop, atrStop)

f_add_signal_line(float price, color lineColor, string textValue) =>
    int endBar = bar_index + signalDrawingBars
    line newLine = line.new(bar_index, price, endBar, price, color=color.new(lineColor, 10), width=1, style=line.style_solid)
    label newLabel = label.new(endBar, price, textValue, style=label.style_label_left, color=color.new(lineColor, 5), textcolor=color.white, size=size.tiny)
    array.push(signalLines, newLine)
    array.push(signalLabels, newLabel)

f_draw_signal(bool isBull, int scoreValue) =>
    color lineColor = isBull ? buyColor : sellColor
    float entry = close
    float stop = f_stop_price(isBull)
    float risk = math.max(math.abs(entry - stop), syminfo.mintick)
    float targetOne = isBull ? entry + risk * targetOneR : entry - risk * targetOneR
    float targetTwo = isBull ? entry + risk * targetTwoR : entry - risk * targetTwoR
    if showSignalLabels
        string textValue = isBull ? "BUY\n" + str.tostring(scoreValue) + "/9" : "SELL\n" + str.tostring(scoreValue) + "/9"
        label signalLabel = label.new(bar_index, isBull ? low : high, textValue, style=isBull ? label.style_label_up : label.style_label_down, color=color.new(lineColor, 0), textcolor=color.white, size=size.small)
        array.push(signalLabels, signalLabel)
    if showSignalLevels
        f_add_signal_line(entry, lineColor, isBull ? "Buy entry" : "Sell entry")
        f_add_signal_line(stop, color.gray, "Stop guide")
        f_add_signal_line(targetOne, lineColor, "Target 1R")
        f_add_signal_line(targetTwo, lineColor, "Target 2R")
    f_trim_signals()
